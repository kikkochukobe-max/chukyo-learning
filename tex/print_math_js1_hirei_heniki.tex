\newcommand{\PageTwo}{%
  \PH{exc}{例題}{変域の求め方}{\DateR}
  \Ins{グラフに $x$ の変域（青）をとり、その部分のグラフ（赤い太線）を $y$ 軸に写すと、$y$ の変域（赤）が見えます。}
  \Ex{\GraphCore{4.2mm}{1}{2}{1}{-1}{1}{3}{1}}{%
    {\bfseries\fontsize{9.3}{13}\selectfont\Tag{exc}{例題1}$y=2x$ で、$x$ の変域が $-1\leqq x\leqq 3$ のとき、$y$ の変域を求めなさい。}
    \begin{enumerate}
      \item グラフをかく：原点と点 $(1,\,2)$ を通る直線。
      \item $x$ の変域のはしの $y$ を求める。\\ $x=-1$ のとき $y=-2$、\ $x=3$ のとき $y=6$
      \item 太い部分を横に見て、$y$ 軸に写す。
    \end{enumerate}
    \ExAns{$-2\leqq y\leqq 6$}}
  \Sep
  \Ex{\GraphCore{4.2mm}{1}{-2}{1}{-2}{0}{1}{1}}{%
    {\bfseries\fontsize{9.3}{13}\selectfont\Tag{exc}{例題2}$y=-2x$ で、$x$ の変域が $-2<x\leqq 1$ のとき、$y$ の変域を求めなさい。}
    \begin{enumerate}
      \item グラフは右下がり：原点と点 $(1,\,-2)$ を通る直線。
      \item $x=-2$ のとき $y=4$（$<$ なので、ふくまない ○）\\ $x=1$ のとき $y=-2$（$\leqq$ なので、ふくむ ●）
      \item 右下がりのときは、$x$ が大きいはしで $y$ が小さくなる。
    \end{enumerate}
    \ExAns{$-2\leqq y<4$}
    \Tip{$\leqq$ のはしは ●（ふくむ）、$<$ のはしは ○（ふくまない）。\\
         $a$ が負のときは $y$ の大小が入れかわるので、小さいほうから書きなおそう。}}
  \Sep
  \Ex{\GraphCore{4.2mm}{1}{-1}{1}{-2}{1}{4}{1}}{%
    {\bfseries\fontsize{9.3}{13}\selectfont\Tag{appc}{例題3 応用}$y=ax$ で、$x$ の変域が $-2\leqq x\leqq 4$ のとき、$y$ の変域は $-4\leqq y\leqq 2$ です。$a$ の値を求めなさい。}
    \begin{enumerate}
      \item 右上がり（$a>0$）なら $x=4$ で $y=2$\\
            → $a=\frac{1}{2}$、すると $x=-2$ で $y=-1$ … {\color{shu}\bfseries ×}（$-4$ にならない）
      \item 右下がり（$a<0$）なら $x=4$ で $y=-4$\\
            → $a=-1$、すると $x=-2$ で $y=2$ … {\color{exc}\bfseries ○}
    \end{enumerate}
    \ExAns{$a=-1$}
    \Tip{コツ：$x$ のはしのうち原点から遠いほう（$4$）が、$y$ のはしの原点から遠いほう（$-4$）に対応する。だから右下がり！}}
  \PN{2}}
